## Params

### Messages

Define params & messages

```ts
import { FormErrorMessages } from './form-error-message.interface';

interface YourParams {
	required: object;
	minlength: { requiredLength: number; actualLength: number };
	pattern: { requiredPattern: string; actualValue: string };
}

const formErrorMessages: FormErrorMessages<YourParams> = {
	required: () => 'Field is required.',
	minlength: ({ requiredLength, actualLength }) => `Minimum ${requiredLength} characters (current ${actualLength}).`,
	pattern: {
		filter: (_, control) => control.dirty || control.touched,
		message: ({ requiredPattern }) => `Invalid format. Expected pattern: ${requiredPattern}`,
	},
};
```

Provide globally

```ts
{
	providers: [{ provide: FORM_ERROR_MESSAGES, useValue: formErrorMessages }];
}
```

Use in template

```html
<input formControlName="password" uiFormError [uiFormErrorMessages]="formErrorMessages" />
```
